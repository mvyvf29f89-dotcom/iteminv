local function getItemMutation(itemName)

	if not itemName then
		return nil, nil
	end

	local name =
		tostring(itemName)

	if name:sub(1, 6) == "Shiny " then
		return "Shiny", name:sub(7)
	end

	if name:sub(1, 5) == "Gold " then
		return "Gold", name:sub(6)
	end

	if name:sub(1, 8) == "Diamond " then
		return "Diamond", name:sub(9)
	end

	if name:sub(1, 8) == "Rainbow " then
		return "Rainbow", name:sub(9)
	end

	if name:sub(1, 5) == "Void " then
		return "Void", name:sub(6)
	end

	return nil, name

end

local function getBaseItemName(itemName)

	local _, baseName =
		getItemMutation(itemName)

	return baseName

end

local function findItemModel(itemName, rarity)

	if not itemName then
		return nil
	end

	local mutation, baseName =
		getItemMutation(itemName)

	if not baseName
		or baseName == "" then

		return nil
	end

	if mutation then

		local mutationFolderName =
			MUTATION_FOLDERS[mutation]

		local mutationFolder =
			mutationFolderName
			and workspace:FindFirstChild(
				mutationFolderName,
				true
			)

		if mutationFolder then

			if rarity
				and tostring(rarity) ~= "" then

				local rarityFolderName =
					tostring(rarity)
					.. " normal items"

				local rarityFolder =
					mutationFolder:FindFirstChild(
						rarityFolderName,
						true
					)

				if rarityFolder then

					if (
						rarityFolder:IsA("Model")
							or rarityFolder:IsA("BasePart")
						)
							and (
								rarityFolder.Name == baseName
								or rarityFolder.Name == itemName
							) then

						return rarityFolder

					end

					for _, descendant in ipairs(
						rarityFolder:GetDescendants()
						) do

						if (
							descendant.Name == baseName
								or descendant.Name == itemName
							)
								and (
									descendant:IsA("Model")
									or descendant:IsA("BasePart")
								) then

							return descendant

						end

					end

				end

			end

			for _, descendant in ipairs(
				mutationFolder:GetDescendants()
				) do

				if (
					descendant.Name == baseName
						or descendant.Name == itemName
					)
						and (
							descendant:IsA("Model")
							or descendant:IsA("BasePart")
						) then

					return descendant

				end

			end

		end

	end

	if rarity
		and tostring(rarity) ~= "" then

		local rarityFolderName =
			tostring(rarity)
			.. " normal items"

		local folder =
			workspace:FindFirstChild(
				rarityFolderName,
				true
			)

		if folder then

			if (
				folder:IsA("Model")
					or folder:IsA("BasePart")
				)
					and folder.Name == baseName then

				return folder

			end

			for _, descendant in ipairs(
				folder:GetDescendants()
				) do

				if descendant.Name == baseName
					and (
						descendant:IsA("Model")
							or descendant:IsA("BasePart")
					) then

					return descendant

				end

			end

		end

	end

	for _, folderName in ipairs(
		ITEM_RARITY_FOLDERS
		) do

		local folder =
			workspace:FindFirstChild(
				folderName,
				true
			)

		if folder then

			if (
				folder:IsA("Model")
					or folder:IsA("BasePart")
				)
					and folder.Name == baseName then

				return folder

			end

			for _, descendant in ipairs(
				folder:GetDescendants()
				) do

				if descendant.Name == baseName
					and (
						descendant:IsA("Model")
							or descendant:IsA("BasePart")
					) then

					return descendant

				end

			end

		end

	end

	return nil

end

	end

-- =========================================================
-- REFRESH INVENTORY
-- =========================================================

local function refreshItemsPanel()

	clearChildrenOf(
		itemScroll
	)

	local entries = {}

	for name, data in pairs(
		localInventory
		) do

		if data
			and data.quantity
			and data.quantity > 0 then

			table.insert(
				entries,
				{
					name = name,
					data = data
				}
			)

		end

	end

	if #entries == 0 then

		local empty =
			Instance.new("TextLabel")

		empty.Size =
			UDim2.new(
				1,
				0,
				0,
				60
			)

		empty.BackgroundTransparency =
			1

		empty.Text =
			"No items yet. Crack an egg!"

		empty.TextColor3 =
			Color3.fromRGB(
				180,
				180,
				190
			)

		empty.TextSize =
			16

		empty.Font =
			Enum.Font.FredokaOne

		empty.TextXAlignment =
			Enum.TextXAlignment.Center

		empty.ZIndex =
			22

		empty.Parent =
			itemScroll

		return

	end

	table.sort(
		entries,
		function(a, b)

			local ra =
				table.find(
					ItemConfig.RARITIES,
					a.data.rarity
				)
				or 0

			local rb =
				table.find(
					ItemConfig.RARITIES,
					b.data.rarity
				)
				or 0

			if ra == rb then
				return a.name < b.name
			end

			return ra > rb

		end
	)

	for _, entry in ipairs(
		entries
		) do

		buildItemCard(
			entry.name,
			entry.data
		)

	end

end

local function updateCrackButton()

	local ownedEgg =
		player:FindFirstChild(
			"OwnedEgg"
		)

	local hasEgg =
		false

	if ownedEgg
		and ownedEgg.Value ~= "" then

		hasEgg =
			true

	end

	if not hasEgg then

		local character =
			player.Character

		if character then

			for _, child in ipairs(
				character:GetChildren()
				) do

				if child:IsA("Tool") then

					hasEgg =
						true

					break

				end

			end

		end

	end

	if not hasEgg then

		local backpack =
			player:FindFirstChildOfClass(
				"Backpack"
			)

		if backpack then

			for _, child in ipairs(
				backpack:GetChildren()
				) do

				if child:IsA("Tool") then

					hasEgg =
						true

					break

				end

			end

		end

	end

	autoButton.Visible =
		hasEgg

	if hasEgg then

		crackButton.Visible =
			true

		if not isRevealing
			and not isCooldown then

			crackButton.Text =
				"CRACK EGG"

		end

	else

		crackButton.Visible =
			false

	end

end

	end

	CrackRequestEvent:
		FireServer()

end

local function getGroundY(
	hrpPos,
	ignore
)

	local params =
		RaycastParams.new()

	params.FilterDescendantsInstances =
		ignore

	params.FilterType =
		Enum.RaycastFilterType.Exclude

	local result =
		workspace:Raycast(
			hrpPos
			+ Vector3.new(
				0,
				60,
				0
			),

			Vector3.new(
				0,
				-250,
				0
			),

			params
		)

	if result then

		return result.Position.Y

	end

	return hrpPos.Y - 3

end

local function createGroundReward(
	itemName,
	rarity,
	landPos
)

	local rewardSource =
		findItemModel(
			itemName,
			rarity
		)

	if not rewardSource then

		warn(
			"[ItemsClient] Missing reward model:",
			itemName
		)

		return nil

	end

	local ok, clone =
		pcall(
			function()

				return ItemViewportUtils
				.cloneAndAnchor(
					rewardSource
				)

			end
		)

	end

	clone.Parent =
		workspace

	clone:PivotTo(
		CFrame.new(
			landPos
		)
	)

	return clone

end

local function cameraKick(
	intensity
)

	local cam =
		workspace.CurrentCamera

	if not cam then
		return
	end

	task.spawn(
		function()

			local origCF =
				cam.CFrame

			local shake =
				0.02
				* intensity

			for i = 1, 3 do

				cam.CFrame =
					origCF
					* CFrame.Angles(

						(math.random() - 0.5)
							* shake,

						(math.random() - 0.5)
							* shake,

						0

					)

				task.wait(
					0.03
				)

				shake =
					shake * 0.6

			end

		end
	)

end

local function createEggClone(
	eggName
)

	if not eggName
		or eggName == "" then

		eggName =
			"Normal Egg"

	end

	local eggModel =
		workspace:FindFirstChild(
			eggName,
			true
		)

	if not eggModel then

		for _, child in ipairs(
			workspace:GetChildren()
			) do

			if child:IsA("MeshPart")
				or child:IsA("Model") then

				if string.find(
						string.lower(
							child.Name
						),
						string.lower(
							eggName
						)
					) then

					eggModel =
						child

					break

				end

			end

		end

	end

	if not eggModel then

		warn(
			"[ItemsClient] Missing egg model:",
			eggName
		)

		return nil

	end

	local clone

	local ok, result =
		pcall(
			function()

				return eggModel:Clone()

			end
		)

	if ok and result then

		clone =
			result

	else

		return nil

	end

	for _, desc in ipairs(
		clone:GetDescendants()
		) do

		if desc:IsA("BasePart") then

			desc.Anchored =
				true

			desc.CanCollide =
				false

		end

	end

	if clone:IsA("BasePart") then

		clone.Anchored =
			true

		clone.CanCollide =
			false

	end

	clone.Parent =
		effectsFolder

	return clone

end

local function popReward(
	rewardClone,
	landPos,
	color,
	direction
)

	if not rewardClone then
		return
	end

	direction =
		direction or 1

	task.spawn(
		function()

			local startTime =
				os.clock()

			local duration =
				0.8

			while os.clock() - startTime < duration do

				local t =
					(os.clock() - startTime)
					/ duration

				local height =
					math.sin(
						t * math.pi
					) * 3

				local scale =
					0.1
					+ 0.9
						* (
							1
							- (1 - t)
								* (1 - t)
						)

				if rewardClone
					and rewardClone.Parent then

					rewardClone:PivotTo(

						CFrame.new(
							landPos
								+ Vector3.new(
									0,
									height,
									0
								)
						)
							* CFrame.Angles(
								0,
								t
									* math.rad(
										360
									)
									* direction,
								0
							)

					)

					pcall(
						function()

							rewardClone:ScaleTo(
								scale
							)

						end
					)

				end

				task.wait(
					0.02
				)

			end

			if rewardClone
				and rewardClone.Parent then

				rewardClone:PivotTo(
					CFrame.new(
						landPos
					)
				)

				pcall(
						function()

							rewardClone:ScaleTo(
								1
							)

						end
					)

			end

		end
	)

end

local function throwEgg(
	eggClone,
	startPos,
	landPos
)

	if not eggClone then

		task.wait(
			0.5
		)

		return

	end

	local throwDist =
		(
			startPos
			- landPos
		).Magnitude

	local arcHeight =
		math.clamp(
			throwDist * 0.55,
			4,
			14
		)

	local upDir =
		Vector3.new(
			0,
			1,
			0
		)

	local throwDir =
		landPos
	- startPos

	throwDir =
		Vector3.new(
			throwDir.X,
			0,
			throwDir.Z
		)

	if throwDir.Magnitude
		> 0.01 then

		throwDir =
			throwDir.Unit

	else

		throwDir =
			Vector3.new(
				0,
				0,
				-1
			)

	end

	local sideDir =
		upDir:Cross(
			throwDir
		)

	local trailHost =
		Instance.new("Part")

	trailHost.Name =
		"EggTrailHost"

	trailHost.Size =
		Vector3.new(
			0.2,
			0.2,
			0.2
		)

	trailHost.Anchored =
		true

	trailHost.CanCollide =
		false

	trailHost.Transparency =
		1

	trailHost.CFrame =
		CFrame.new(
			startPos
		)

	trailHost.Parent =
		effectsFolder

	local trailAtt =
		Instance.new("Attachment")

	trailAtt.Parent =
		trailHost

	local trailPE =
		Instance.new("ParticleEmitter")

	trailPE.Color =
		ColorSequence.new({

			ColorSequenceKeypoint.new(
				0,
				Color3.fromRGB(
					255,
					235,
					160
				)
			),

			ColorSequenceKeypoint.new(
				0.5,
				Color3.fromRGB(
					255,
					200,
					80
				)

local function collectItemPool()

	local pool = {}

	for _, folderName in ipairs(
		ITEM_RARITY_FOLDERS
		) do

		local folder =
			workspace:FindFirstChild(
				folderName,
				true
			)

		if folder then

			for _, child in ipairs(
				folder:GetChildren()
				) do

				if child:IsA("Model")
					or child:IsA("BasePart") then

					table.insert(
						pool,
						child
					)

				end

			end

		end

	end

	for _, folderName in pairs(
		MUTATION_FOLDERS
		) do

		local folder =
			workspace:FindFirstChild(
				folderName,
				true
			)

		if folder then

			for _, child in ipairs(
				folder:GetChildren()
				) do

				if child:IsA("Model")
					or child:IsA("BasePart") then

					table.insert(
						pool,
						child
					)

				end

			end

		end

	end

	return pool

end

-- =========================================================
-- ANCHOR CLONE
-- =========================================================

local function anchorClone(
	clone
)

	if not clone then
		return
	end

	for _, descendant in ipairs(
		clone:GetDescendants()
		) do

		if descendant:IsA(
			"BasePart"
			) then

			descendant.Anchored =
				true

			descendant.CanCollide =
				false

		end

	end

end

-- =========================================================

local function playItemReel(
	winnerName
)

	pcall(
		function()

			local cam =
				workspace.CurrentCamera

			if not cam then
				return
			end

			local winnerModel =
				findItemModel(
					winnerName
				)

			local pool =
				collectItemPool()

			local reelItems =
				{}

			if #pool == 0
				and not winnerModel then

				return

			end

			local camCF =
				cam.CFrame

			local rowDist =
				7

			local spacing =
				2.4

			local reelCount =
				10

			local seq =
				{}

			for i = 1, reelCount do

				if #pool > 0 then

					seq[#seq + 1] =
						pool[
					math.random(
						#pool
					)
					]

				end

			end

			if winnerModel then

				seq[#seq + 1] =
					winnerModel

			elseif #pool > 0 then

				seq[#seq + 1] =
					pool[
				math.random(
					#pool
				)
				]

			end

			local total =
				#seq

			if total == 0 then
				return
			end

			for i, model in ipairs(
				seq
				) do

				local clone =
					model:Clone()

				anchorClone(
					clone
				)

				pcall(
					function()

						clone:ScaleTo(
							1.3
						)

					end
				)

				clone:PivotTo(

					camCF
						* CFrame.new(

							(i - total)
							* spacing
							- total
							* spacing,

							0,

							-rowDist

						)

				)

				clone.Parent =
					workspace

				reelItems[#reelItems + 1] =
					clone

			end

			local duration =
				1.9

			local startOffset =
				-(total)
				* spacing
			- 3

			local t0 =
				os.clock()

			while true do

				local t =
					(
						os.clock()
						- t0
					)
					/ duration

				if t >= 1 then
					break
				end

				local eased =
					1
				- (1 - t)
					* (1 - t)
					* (1 - t)

				local off =
					startOffset
					+ (
						0
						- startOffset
					)
					* eased

				local spin =
					os.clock()
					* 5

				for i, clone in ipairs(
					reelItems
					) do

					local baseX =
						(i - total)
						* spacing

					clone:PivotTo(

						camCF
							* CFrame.new(

								baseX
								+ off,

								0,

								-rowDist

							)
							* CFrame.Angles(
								0,
								spin,
								0
							)

					)

				end

				task.wait(
					0.016
				)

			end

			for _, clone in ipairs(
				reelItems
				) do

				if clone
					and clone.Parent then

					clone:Destroy()

				end

			end

		end
	)

end

local function abbreviateNumber(value)

	value =
		math.floor(
			tonumber(value)
			or 0
		)

	local negative =
		value < 0

	value =
		math.abs(
			value
		)

	if value < 1000 then

		return
			(negative and "-" or "")
			.. tostring(
				value
			)

	end

	local tier =
		math.floor(
			math.log10(
				value
			) / 3
		)

	if tier >= #SUFFIXES then

		local exp =
			math.floor(
				math.log10(
					value
				)
			)

		local mantissa =
			value
			/ (
				10 ^ exp
			)

		return
			(negative and "-" or "")
			.. string.format(
				"%.2fe%d",
				mantissa,
				exp
			)

	end

	local suffix =
		SUFFIXES[
	tier + 1
	]

	local scaled =
		value
		/ (
			10 ^ (
				tier * 3
			)
		)

	local formatted

	if scaled >= 100 then

		formatted =
			string.format(
				"%.1f",
				scaled
			)

	elseif scaled >= 10 then

		formatted =
			string.format(
				"%.2f",
				scaled
			)

	else

		formatted =
			string.format(
				"%.2f",
				scaled
			)

	end

	formatted =
		formatted:gsub(
			"(%d)%.(%d%d+)$",
			function(
				whole,
				dec
			)

				dec =
				dec:gsub(
					"0+$",
					""
				)

				if dec == "" then
					return whole
				end

				return
				whole
				.. "."
				.. dec

			end
		)

	return
		(negative and "-" or "")
		.. formatted
		.. suffix

end
